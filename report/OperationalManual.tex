\subsection{Нэмэлт boundary ба error тест}
\begin{lstlisting}[language=bash]
./run.sh --headless -n 0                      # хоосон массив
./run.sh --headless -n 1                      # нэг элемент
./run.sh --headless -n 10 -t 1000             # threshold > n -> ганц навч
./run.sh --headless -n 100000 -d ALL_EQUAL -a quick   # давхардсан утга
./run.sh --headless -n 100000 -d SORTED    -a quick   # аль хэдийн эрэмбэлэгдсэн
./run.sh --headless -i "1, abc, 3"            # алдаатай оролт
./run.sh --headless -n 1000 -t 0              # хүчингүй threshold
\end{lstlisting}

\subsection{Хүлээгдсэн үр дүн}
\begin{itemize}[nosep,leftmargin=*]
  \item \texttt{./test.sh} $\Rightarrow$ \texttt{=== ДҮН: 58 давсан, 0 унасан ===},
        exit code \textbf{0}.
  \item Аливаа \texttt{--headless} ажиллуулалтад \texttt{Зөв эсэх : ЗӨВ}.
        Буруу бол exit code \textbf{2}.
  \item Хүчингүй оролтод Java-гийн stack trace биш, \emph{монгол хэлээр
        ойлгомжтой} мессеж гарна.
  \item \texttt{--bench} $\Rightarrow$ CSV хэлбэрийн гурван хүснэгт.
        Хоёр цөмт машин дээр threshold $\ge 50$ үед хурдасгалт $\approx 2.0$.
\end{itemize}

\subsection{Known issues}
\begin{enumerate}[nosep,leftmargin=*]
  \item \textbf{Модны бүртгэл хурдад нөлөөлнө.} Зангилаа бүрийг бүртгэх нь
        нэмэлт зардал үүсгэдэг тул \texttt{--bench} нь
        \texttt{--no-trace} горимоор хэмждэг. UI-д бүртгэл \emph{үргэлж} асаалттай
        байдаг тул тэнд харагдах хугацааг цэвэр хэмжилт гэж үзэж болохгүй.
  \item \textbf{Маш том мод.} $n = 10^6$, threshold $= 10$ үед 200\,000 орчим
        зангилаа үүсч, зурах нь удаашрана. Дүрслэлд $n \le 5000$ хооронд
        ажиллуулахыг зөвлөнө; хурдны хэмжилтэд \texttt{--bench} ашиглана.
  \item \textbf{Зангилааны утгуудыг зөвхөн 48-аас бага мужид хадгална}
        (санах ойн хязгаарлалт). Түүнээс том муж дээр дарвал утгууд харагдахгүй.
  \item \textbf{Хурдасгалт нь машинаас хамаарна.} Хөгжүүлэлтийн орчин
        \textbf{хоёр} цөмтэй байсан тул дээд хурдасгалт $\approx 2$. Олон цөмтэй
        машин дээр илүү өндөр гарна.
  \item \textbf{\texttt{Integer} объектын массив.} Сурах бичиг
        \texttt{Comparable} интерфейсийг шаарддаг тул \texttt{int[]} биш
        \texttt{Integer[]} ашигласан. Энэ нь boxing-ийн улмаас анхдагч төрлөөс
        удаан боловч даалгаврын шаардлагыг хангана.
\end{enumerate}
